1//! Definition of all client and request configuration options.
2//!
3//! Individual options are separated out into multiple types. Each type acts
4//! both as a "field name" and the value of that option.
5
6// Options are implemented as structs of various kinds that can be "applied" to
7// a curl easy handle. This helps to reduce code duplication as there are many
8// supported options, and also helps avoid having a massive function that does
9// all the configuring.
10//
11// When adding new config options, remember to add methods for setting the
12// option both in HttpClientBuilder and RequestBuilderExt. In addition, be sure
13// to update the client code to apply the option when configuring an easy
14// handle.
15

32/// Provides additional methods when building a request for configuring various
33/// execution-related options on how the request should be sent.
34///
35/// This trait can be used to either configure requests individually by invoking
36/// them on an [`http::request::Builder`], or to configure the default settings
37/// for an [`HttpClient`](crate::HttpClient) by invoking them on an
38/// [`HttpClientBuilder`](crate::HttpClientBuilder).
39///
40/// This trait is sealed and cannot be implemented for types outside of Isahc.
41pub trait Configurable: request::WithRequestConfig {
42    /// Specify a maximum amount of time that a complete request/response cycle
43    /// is allowed to take before being aborted. This includes DNS resolution,
44    /// connecting to the server, writing the request, and reading the response.
45    ///
46    /// Since response bodies are streamed, you will likely receive a
47    /// [`Response`](crate::http::Response) before the response body stream has
48    /// been fully consumed. This means that the configured timeout will still
49    /// be active for that request, and if it expires, further attempts to read
50    /// from the stream will return a [`TimedOut`](std::io::ErrorKind::TimedOut)
51    /// I/O error.
52    ///
53    /// This also means that if you receive a response with a body but do not
54    /// immediately start reading from it, then the timeout timer will still be
55    /// active and may expire before you even attempt to read the body. Keep
56    /// this in mind when consuming responses and consider handling the response
57    /// body right after you receive it if you are using this option.
58    ///
59    /// If not set, no timeout will be enforced.
60    ///
61    /// # Examples
62    ///
63    /// ```no_run
64    /// use isahc::{prelude::*, Request};
65    /// use std::time::Duration;
66    ///
67    /// // This page is too slow and won't respond in time.
68    /// let response = Request::get("https://httpbin.org/delay/10")
69    ///     .timeout(Duration::from_secs(5))
70    ///     .body(())?
71    ///     .send()
72    ///     .expect_err("page should time out");
73    /// # Ok::<(), isahc::Error>(())
74    /// ```
75    #[must_use = "builders have no effect if unused"]
76    fn timeout(self, timeout: Duration) -> Self {
77        self.with_config(move |config| {
78            config.timeout = Some(timeout);
79        })
80    }

209    /// Configure the use of the `Expect` request header when sending request
210    /// bodies with HTTP/1.1.
211    ///
212    /// By default, when sending requests containing a body of large or unknown
213    /// length over HTTP/1.1, Isahc will send the request headers first without
214    /// the body and wait for the server to respond with a 100 (Continue) status
215    /// code, as defined by [RFC 7231, Section
216    /// 5.1.1](https://datatracker.ietf.org/doc/html/rfc7231#section-5.1.1).
217    /// This gives the opportunity for the server to reject the response without
218    /// needing to first transmit the request body over the network, if the body
219    /// contents are not necessary for the server to determine an appropriate
220    /// response.
221    ///
222    /// For servers that do not support this behavior and instead simply wait
223    /// for the request body without responding with a 100 (Continue), there is
224    /// a limited timeout before the response body is sent anyway without
225    /// confirmation. The default timeout is 1 second, but this can be
226    /// configured.
227    ///
228    /// The `Expect` behavior can also be disabled entirely.
229    ///
230    /// This configuration only takes effect when using HTTP/1.1.
231    ///
232    /// # Examples
233    ///
234    /// ```
235    /// use std::time::Duration;
236    /// use isahc::{
237    ///     config::ExpectContinue,
238    ///     prelude::*,
239    ///     HttpClient,
240    /// };
241    ///
242    /// // Use the default behavior (enabled).
243    /// let client = HttpClient::builder()
244    ///     .expect_continue(ExpectContinue::enabled())
245    ///     // or equivalently...
246    ///     .expect_continue(ExpectContinue::default())
247    ///     // or equivalently...
248    ///     .expect_continue(true)
249    ///     .build()?;
250    ///
251    /// // Customize the timeout if the server doesn't respond with a 100
252    /// // (Continue) status.
253    /// let client = HttpClient::builder()
254    ///     .expect_continue(ExpectContinue::timeout(Duration::from_millis(200)))
255    ///     // or equivalently...
256    ///     .expect_continue(Duration::from_millis(200))
257    ///     .build()?;
258    ///
259    /// // Disable the Expect header entirely.
260    /// let client = HttpClient::builder()
261    ///     .expect_continue(ExpectContinue::disabled())
262    ///     // or equivalently...
263    ///     .expect_continue(false)
264    ///     .build()?;
265    /// # Ok::<(), isahc::Error>(())
266    /// ```
267    #[must_use = "builders have no effect if unused"]
268    fn expect_continue<T>(self, expect: T) -> Self
269    where
270        T: Into<ExpectContinue>,
271    {
272        self.with_config(move |config| {
273            config.expect_continue = Some(expect.into());
274        })
275    }

333    /// Configure which network interface local sockets should be bound to for sending
334    /// network traffic from.
335    ///
336    /// Note that interface selection is not performed until you actually
337    /// attempt to send a request, so the validity of your selection is not
338    /// checked here.
339    ///
340    /// # Examples
341    ///
342    /// Bind to an IP address.
343    ///
344    /// ```
345    /// use isahc::{prelude::*, HttpClient};
346    /// use std::net::IpAddr;
347    ///
348    /// let client = HttpClient::builder()
349    ///     .interface(IpAddr::from([192, 168, 1, 2]))
350    ///     .build()?;
351    /// # Ok::<(), isahc::Error>(())
352    /// ```
353    ///
354    /// Bind to an interface by name (not supported on Windows).
355    ///
356    /// ```
357    /// # #[cfg(unix)] {
358    /// use isahc::{prelude::*, net::interface::Name, HttpClient};
359    ///
360    /// let client = HttpClient::builder()
361    ///     .interface(Name("eth0"))
362    ///     .build()?;
363    /// # }
364    /// # Ok::<(), isahc::Error>(())
365    /// ```
366    ///
367    /// Bind to an interface by both name and IP address.
368    ///
369    /// ```
370    /// # #[cfg(unix)] {
371    /// use isahc::{prelude::*, net::interface::Name, HttpClient};
372    /// use std::net::IpAddr;
373    ///
374    /// let client = HttpClient::builder()
375    ///     .interface((Name("eth0"), IpAddr::from([192, 168, 1, 2])))
376    ///     .build()?;
377    /// # }
378    /// # Ok::<(), isahc::Error>(())
379    /// ```
380    ///
381    /// Reset to using whatever interface the TCP stack finds suitable (the
382    /// default).
383    ///
384    /// ```
385    /// use isahc::{prelude::*, net::interface::Any, Request};
386    ///
387    /// let request = Request::get("https://example.org")
388    ///     .interface(Any)
389    ///     .body(())?;
390    /// # Ok::<(), isahc::Error>(())
391    /// ```
392    #[must_use = "builders have no effect if unused"]
393    fn interface<I>(self, interface: I) -> Self
394    where
395        I: interface::Selector,
396    {
397        self.with_config(move |config| {
398            config.interface = Some(interface.into());
399        })
400    }

464    /// Set a proxy to use for requests.
465    ///
466    /// The proxy protocol is specified by the URI scheme.
467    ///
468    /// - **`http`**: Proxy. Default when no scheme is specified.
469    /// - **`https`**: HTTPS Proxy. (Added in 7.52.0 for OpenSSL, GnuTLS and
470    ///   NSS)
471    /// - **`socks4`**: SOCKS4 Proxy.
472    /// - **`socks4a`**: SOCKS4a Proxy. Proxy resolves URL hostname.
473    /// - **`socks5`**: SOCKS5 Proxy.
474    /// - **`socks5h`**: SOCKS5 Proxy. Proxy resolves URL hostname.
475    ///
476    /// By default no proxy will be used, unless one is specified in either the
477    /// `http_proxy` or `https_proxy` environment variables.
478    ///
479    /// Setting to `None` explicitly disables the use of a proxy.
480    ///
481    /// # Examples
482    ///
483    /// Using `http://proxy:80` as a proxy:
484    ///
485    /// ```
486    /// use isahc::{prelude::*, HttpClient};
487    ///
488    /// let client = HttpClient::builder()
489    ///     .proxy(Some("http://proxy:80".parse()?))
490    ///     .build()?;
491    /// # Ok::<(), Box<dyn std::error::Error>>(())
492    /// ```
493    ///
494    /// Explicitly disable the use of a proxy:
495    ///
496    /// ```
497    /// use isahc::{prelude::*, HttpClient};
498    ///
499    /// let client = HttpClient::builder()
500    ///     .proxy(None)
501    ///     .build()?;
502    /// # Ok::<(), Box<dyn std::error::Error>>(())
503    /// ```
504    #[must_use = "builders have no effect if unused"]
505    fn proxy(self, proxy: impl Into<Option<http::Uri>>) -> Self {
506        self.with_config(move |config| {
507            config.proxy = Some(proxy.into());
508        })
509    }

661    ///
662    /// By default, the same TLS configuration is used for validating all
663    /// SSL/TLS connections, but this method allows you to use separate
664    /// configuration specifically for proxy server connections.
665    ///
666    /// # Warning
667    ///
668    /// You should think very carefully before using this method. Using *any*
669    /// options that alter how certificates are validated can introduce
670    /// significant security vulnerabilities.
671    ///
672    /// # Examples
673    ///
674    /// ```no_run
675    /// use isahc::{prelude::*, tls::{Identity, PrivateKey, TlsConfig, TrustStore}, Request};
676    ///
677    /// let response = Request::get("https://example.org")
678    ///     .proxy_tls_config(TlsConfig::builder()
679    ///         .trust_store(TrustStore::native())
680    ///         .identity(Identity::from_pem_file(
681    ///             "client.pem",
682    ///             Some(PrivateKey::from_pem_file(
683    ///                 "key.pem",
684    ///                 String::from("secret")
685    ///             ))
686    ///         ))
687    ///         .build())
688    ///     .body(())?
689    ///     .send()?;
690    /// # Ok::<(), isahc::Error>(())
691    /// ```
692    ///
693    /// ```
694    /// use isahc::{prelude::*, tls::TlsConfig, HttpClient};
695    ///
696    /// # #[cfg(feature = "tls-insecure")]
697    /// let client = HttpClient::builder()
698    ///     .proxy_tls_config(TlsConfig::builder()
699    ///         .danger_accept_invalid_certs(true)
700    ///         .danger_accept_revoked_certs(true)
701    ///         .build())
702    ///     .build()?;
703    /// # Ok::<(), isahc::Error>(())
704    /// ```
705    #[cfg(feature = "tls")]
706    #[must_use = "builders have no effect if unused"]
707    fn proxy_tls_config(self, tls_config: crate::tls::TlsConfig) -> Self {
708        self.with_config(move |config| {
709            config.proxy_tls_config = Some(tls_config);
710        })
711    }
712
713    /// Enable or disable sending HTTP header names in Title-Case instead of
714    /// lowercase form.
715    ///
716    /// This option only affects user-supplied headers and does not affect
717    /// low-level headers that are automatically supplied for HTTP protocol
718    /// details, such as `Connection` and `Host` (unless you override such a
719    /// header yourself).
720    ///
721    /// This option has no effect when using HTTP/2 or newer where headers are
722    /// required to be lowercase.
723    #[must_use = "builders have no effect if unused"]
724    fn title_case_headers(self, enable: bool) -> Self {
725        self.with_config(move |config| {
726            config.title_case_headers = Some(enable);
727        })
728    }
729
730    /// Enable or disable comprehensive per-request metrics collection.
731    ///
732    /// When enabled, detailed timing metrics will be tracked while a request is
733    /// in progress, such as bytes sent and received, estimated size, DNS lookup
734    /// time, etc. For a complete list of the available metrics that can be
735    /// inspected, see the [`Metrics`](crate::Metrics) documentation.
736    ///
737    /// When enabled, to access a view of the current metrics values you can use
738    /// [`ResponseExt::metrics`](crate::ResponseExt::metrics).
739    ///
740    /// While effort is taken to optimize hot code in metrics collection, it is
741    /// likely that enabling it will have a small effect on overall throughput.
742    /// Disabling metrics may be necessary for absolute peak performance.
743    ///
744    /// By default metrics are disabled.
745    #[must_use = "builders have no effect if unused"]
746    fn metrics(self, enable: bool) -> Self {
747        self.with_config(move |config| {
748            config.enable_metrics = Some(enable);
749        })
750    }

753/// A strategy for selecting what HTTP versions should be used when
754/// communicating with a server.
755///
756/// You can set a version negotiation strategy on a given request or on a client
757/// with [`Configurable::version_negotiation`].
758///
759/// Attempting to use an HTTP version without client-side support at runtime
760/// will result in an error. For example, using the system libcurl on an old
761/// machine may not have an HTTP/2 implementation. Using static linking and the
762/// [`http2`](../index.html#http2) crate feature can help guarantee that HTTP/2
763/// will be available to use.
764#[derive(Clone, Debug)]
765pub struct VersionNegotiation(VersionNegotiationInner);
766
767#[derive(Clone, Copy, Debug)]
768enum VersionNegotiationInner {
769    LatestCompatible,
770    Strict(curl::easy::HttpVersion),
771}
772
773impl Default for VersionNegotiation {
774    fn default() -> Self {
775        Self::latest_compatible()
776    }
777}
778
779impl VersionNegotiation {
780    /// Always prefer the latest supported version announced by the server,
781    /// falling back to older versions if not explicitly listed as supported.
782    /// This is the default.
783    ///
784    /// Secure connections will begin with a TLS handshake, after which the
785    /// highest supported HTTP version listed by the server via ALPN will be
786    /// used. Once connected, additional upgrades to newer versions may also
787    /// occur if the server lists support for it. In the future, headers such as
788    /// `Alt-Svc` will be used.
789    ///
790    /// Insecure connections always use HTTP/1.x since there is no standard
791    /// mechanism for a server to declare support for insecure HTTP versions,
792    /// and only HTTP/0.9, HTTP/1.x, and HTTP/2 support insecure transfers.
793    pub const fn latest_compatible() -> Self {
794        Self(VersionNegotiationInner::LatestCompatible)
795    }
796
797    /// Connect via HTTP/1.0 and do not attempt to use a higher version.
798    pub const fn http10() -> Self {
799        Self(VersionNegotiationInner::Strict(
800            curl::easy::HttpVersion::V10,
801        ))
802    }
803
804    /// Connect via HTTP/1.1 and do not attempt to use a higher version.
805    pub const fn http11() -> Self {
806        Self(VersionNegotiationInner::Strict(
807            curl::easy::HttpVersion::V11,
808        ))
809    }
810
811    /// Connect via HTTP/2. Failure to connect will not fall back to old
812    /// versions, unless HTTP/1.1 is negotiated via TLS ALPN before the session
813    /// begins.
814    ///
815    /// If HTTP/2 support is not compiled in, then using this strategy will
816    /// always result in an error.
817    ///
818    /// This strategy is often referred to as [HTTP/2 with Prior
819    /// Knowledge](https://http2.github.io/http2-spec/#known-http).
820    pub const fn http2() -> Self {
821        Self(VersionNegotiationInner::Strict(
822            curl::easy::HttpVersion::V2PriorKnowledge,
823        ))
824    }
825
826    /// Connect via HTTP/3. Failure to connect will not fall back to old
827    /// versions.
828    pub const fn http3() -> Self {
829        Self(VersionNegotiationInner::Strict(curl::easy::HttpVersion::V3))
830    }
831}
832
833impl SetOpt for VersionNegotiation {
834    fn set_opt(&self, easy: &mut EasyHandle) -> Result<(), SetOptError> {
835        match self.0 {
836            VersionNegotiationInner::LatestCompatible => {
837                // If HTTP/2 support is available, this basically the most
838                // lenient way of using it. Alt-Svc is used to upgrade to newer
839                // versions, and old versions are used if the server doesn't
840                // list HTTP/2 via ALPN.
841                //
842                // If HTTP/2 is not available, leaving it the default setting is
843                // the ideal behavior.
844                if is_http_version_supported(http::Version::HTTP_2) {
845                    easy.http_version(curl::easy::HttpVersion::V2TLS)
846                        .map_err(Into::into)
847                } else {
848                    Ok(())
849                }
850            }
851            VersionNegotiationInner::Strict(version) => {
852                easy.http_version(version).map_err(Into::into)
853            }
854        }
855    }
856}

858/// Controls the use of the `Expect` request header when sending request bodies
859/// with HTTP/1.1.
860///
861/// By default, when sending requests containing a body of large or unknown
862/// length over HTTP/1.1, Isahc will send the request headers first without the
863/// body and wait for the server to respond with a 100 (Continue) status code,
864/// as defined by [RFC 7231, Section
865/// 5.1.1](https://datatracker.ietf.org/doc/html/rfc7231#section-5.1.1). This
866/// gives the opportunity for the server to reject the response without needing
867/// to first transmit the request body over the network, if the body contents
868/// are not necessary for the server to determine an appropriate response.
869///
870/// For servers that do not support this behavior and instead simply wait for
871/// the request body without responding with a 100 (Continue), there is a
872/// limited timeout before the response body is sent anyway without
873/// confirmation. The default timeout is 1 second, but this can be configured.
874///
875/// The `Expect` behavior can also be disabled entirely.
876///
877/// This configuration only takes effect when using HTTP/1.1.
878#[derive(Clone, Debug)]
879pub struct ExpectContinue {
880    timeout: Option<Duration>,
881}
882
883impl ExpectContinue {
884    /// Enable the use of `Expect` and wait for a 100 (Continue) response with a
885    /// default timeout before sending a request body.
886    pub const fn enabled() -> Self {
887        Self::timeout(Duration::from_secs(1))
888    }
889
890    /// Enable the use of `Expect` and wait for a 100 (Continue) response, up to
891    /// the given timeout, before sending a request body.
892    pub const fn timeout(timeout: Duration) -> Self {
893        Self {
894            timeout: Some(timeout),
895        }
896    }
897
898    /// Disable the use and handling of the `Expect` request header.
899    pub const fn disabled() -> Self {
900        Self { timeout: None }
901    }
902
903    pub(crate) fn is_disabled(&self) -> bool {
904        self.timeout.is_none()
905    }
906}
907
908impl Default for ExpectContinue {
909    fn default() -> Self {
910        Self::enabled()
911    }
912}
913
914impl From<bool> for ExpectContinue {
915    fn from(value: bool) -> Self {
916        if value {
917            Self::enabled()
918        } else {
919            Self::disabled()
920        }
921    }
922}
923
924impl From<Duration> for ExpectContinue {
925    fn from(value: Duration) -> Self {
926        Self::timeout(value)
927    }
928}
929
930impl SetOpt for ExpectContinue {
931    fn set_opt(&self, easy: &mut EasyHandle) -> Result<(), SetOptError> {
932        if let Some(timeout) = self.timeout {
933            easy.expect_100_timeout(timeout)?;
934        }
935
936        Ok(())
937    }
938}
