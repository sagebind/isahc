10/// A specific issuer certificate to use when validating a server's certificate
11/// chain.
12///
13/// Note that curl only supports specifying issuer certificates in PEM format
14/// under the hood, so if your certificate is in DER format, you will need to
15/// convert it to PEM first.
16#[derive(Clone, Debug)]
17pub struct Issuer {
18    pem: PathOrBlob,
19}
20
21impl Issuer {
22    /// Validate the issuer against a PEM-encoded certificate.
23    ///
24    /// The certificate can be supplied as any type which can be referenced as
25    /// contiguous bytes. This could be a simple [`String`], or a pre-parsed PEM
26    /// object that allows access to its underlying bytes. This takes ownership
27    /// of the bytes regardless.
28    ///
29    /// The certificate is not parsed or validated here. If a certificate is
30    /// malformed or the format is not supported by the underlying SSL/TLS
31    /// engine, an error will be returned when attempting to send a request
32    /// using the offending certificate.
33    pub fn from_pem<B: AsRef<[u8]> + Send + Sync + 'static>(pem: B) -> Self {
34        Self {
35            pem: PathOrBlob::Blob(Blob::new(pem)),
36        }
37    }
38
39    /// Validate the issuer against a DER-encoded certificate.
40    ///
41    /// The certificate is copied and first converted into PEM format before
42    /// being stored.
43    ///
44    /// The certificate is not parsed or validated here. If a certificate is
45    /// malformed or the format is not supported by the underlying SSL/TLS
46    /// engine, an error will be returned when attempting to send a request
47    /// using the offending certificate.
48    pub fn copy_from_der<B: AsRef<[u8]>>(der: B) -> Self {
49        let pem =
50            pem_rfc7468::encode_string("CERTIFICATE", Default::default(), der.as_ref()).unwrap();
51
52        Self::from_pem(pem)
53    }
54
55    /// Validate the issuer against a PEM-encoded certificate stored in a file
56    /// with the given path.
57    ///
58    /// The certificate is not parsed or validated here. If a certificate is
59    /// malformed or the format is not supported by the underlying SSL/TLS
60    /// engine, an error will be returned when attempting to send a request
61    /// using the offending certificate.
62    pub fn from_pem_file<P>(path: P) -> Self
63    where
64        P: Into<PathBuf>,
65    {
66        Self {
67            pem: PathOrBlob::Path(path.into()),
68        }
69    }
70}
71
72impl SetOpt for Issuer {
73    fn set_opt(&self, easy: &mut EasyHandle) -> Result<(), SetOptError> {
74        match &self.pem {
75            PathOrBlob::Path(path) => easy.issuer_cert(path)?,
76            PathOrBlob::Blob(blob) => unsafe {
77                easy.setopt_blob_nocopy(CURLOPT_ISSUERCERT_BLOB, blob)?;
78            },
79        }
80
81        Ok(())
82    }
83}
84
85impl SetOptProxy for Issuer {
86    fn set_opt_proxy(&self, easy: &mut EasyHandle) -> Result<(), SetOptError> {
87        match &self.pem {
88            PathOrBlob::Path(path) => easy.proxy_issuer_cert(path)?,
89            PathOrBlob::Blob(blob) => unsafe {
90                easy.setopt_blob_nocopy(CURLOPT_PROXY_ISSUERCERT_BLOB, blob)?;
91            },
92        }
93
94        Ok(())
95    }
